// CollegeService handles business logic for college management.
type CollegeService struct {
	repo *repository.Repository
	db   *database.Database
}

// NewCollegeService creates a new college service.
func NewCollegeService(
	db *database.Database,
	repo *repository.Repository,
) *CollegeService {
	return &CollegeService{
		repo: repo,
		db:   db,
	}
}

// ListColleges retrieves a list of colleges with optional filtering and pagination.
func (s *CollegeService) ListColleges(
	ctx context.Context,
	params models.ListCollegesParams,
) ([]models.College, *models.PaginationMeta, error) {
	// Set defaults for pagination
	limit := 20
	if params.Limit > 0 {
		limit = params.Limit
	}

	offset := 0
	if params.Offset > 0 {
		offset = params.Offset
	}

	// Prepare status filter
	var statusFilter *string
	if params.Status != "" {
		statusFilter = &params.Status
	}

	// Call repository
	colleges, total, err := s.repo.ListColleges(ctx, statusFilter, limit, offset)
	if err != nil {
		return nil, nil, apierrors.New(
			apierrors.InternalError,
			"Failed to retrieve colleges",
		)
	}

	meta := models.NewPaginationMeta(limit, offset, total)

	return colleges, meta, nil
}

// GetCollege retrieves a single college by its ID.
func (s *CollegeService) GetCollege(
	ctx context.Context,
	collegeID uuid.UUID,
) (*models.College, error) {
	college, err := s.repo.GetCollegeByID(ctx, collegeID)
	if err != nil {
		if err == pgx.ErrNoRows {
			return nil, apierrors.New(
				apierrors.NotFound,
				"College not found",
			)
		}
		return nil, apierrors.New(
			apierrors.InternalError,
			"Failed to retrieve college",
		)
	}

	return college, nil
}

// CreateCollege creates a new college with validation.
func (s *CollegeService) CreateCollege(
	ctx context.Context,
	req models.CreateCollegeRequest,
) (*models.College, error) {
	// Validate college code format (alphanumeric, hyphens, underscores)
	if !isValidCollegeCode(req.CollegeCode) {
		return nil, apierrors.New(
			apierrors.ValidationError,
			"College code must contain only alphanumeric characters, hyphens, and underscores",
		)
	}

	// Check for duplicate college code
	existing, err := s.repo.GetCollegeByCode(ctx, req.CollegeCode)
	if err != nil && err != pgx.ErrNoRows {
		return nil, apierrors.New(
			apierrors.InternalError,
			"Failed to check college code availability",
		)
	}
	if existing != nil {
		return nil, apierrors.New(
			apierrors.Conflict,
			"College code already exists",
		)
	}

	// Validate timezone (basic check - should contain valid timezone identifier)
	if !isValidTimezone(req.Timezone) {
		return nil, apierrors.New(
			apierrors.ValidationError,
			"Invalid timezone format",
		)
	}

	// Create college within a transaction
	var college *models.College
	err = s.db.WithPlatformTransaction(ctx, func(ctx context.Context, _ pgx.Tx) error {
		var txErr error
		college, txErr = s.repo.CreateCollege(ctx, req)
		return txErr
	})

	if err != nil {
		return nil, apierrors.New(
			apierrors.InternalError,
			"Failed to create college",
		)
	}

	return college, nil
}

// UpdateCollege updates permitted fields of a college.
func (s *CollegeService) UpdateCollege(
	ctx context.Context,
	collegeID uuid.UUID,
	req models.UpdateCollegeRequest,
) (*models.College, error) {
	// Verify college exists
	_, err := s.repo.GetCollegeByID(ctx, collegeID)
	if err != nil {
		if err == pgx.ErrNoRows {
			return nil, apierrors.New(
				apierrors.NotFound,
				"College not found",
			)
		}
		return nil, apierrors.New(
			apierrors.InternalError,
			"Failed to retrieve college",
		)
	}

	// Validate timezone if provided
	if req.Timezone != nil && !isValidTimezone(*req.Timezone) {
		return nil, apierrors.New(
			apierrors.ValidationError,
			"Invalid timezone format",
		)
	}

	// Validate that at least one field is being updated
	if req.Name == nil && req.City == nil && req.State == nil &&
		req.Timezone == nil && req.ContactEmail == nil {
		return nil, apierrors.New(
			apierrors.ValidationError,
			"At least one field must be provided for update",
		)
	}

	// Update college within a transaction
	var college *models.College
	err = s.db.WithPlatformTransaction(ctx, func(ctx context.Context, _ pgx.Tx) error {
		var txErr error
		college, txErr = s.repo.UpdateCollege(ctx, collegeID, req)
		return txErr
	})

	if err != nil {
		return nil, apierrors.New(
			apierrors.InternalError,
			"Failed to update college",
		)
	}

	return college, nil
}

// UpdateCollegeStatus updates the status of a college with lifecycle tracking.
func (s *CollegeService) UpdateCollegeStatus(
	ctx context.Context,
	collegeID uuid.UUID,
	req models.UpdateCollegeStatusRequest,
) (*models.College, error) {
	// Retrieve current college state
	currentCollege, err := s.repo.GetCollegeByID(ctx, collegeID)
	if err != nil {
		if err == pgx.ErrNoRows {
			return nil, apierrors.New(
				apierrors.NotFound,
				"College not found",
			)
		}
		return nil, apierrors.New(
			apierrors.InternalError,
			"Failed to retrieve college",
		)
	}

	// Validate status transition
	if !isValidStatusTransition(currentCollege.Status, req.Status) {
		return nil, apierrors.New(
			apierrors.ValidationError,
			fmt.Sprintf(
				"Invalid status transition from %s to %s",
				currentCollege.Status,
				req.Status,
			),
		)
	}

	// Require reason for certain status changes
	if requiresReason(req.Status) && (req.Reason == nil || *req.Reason == "") {
		return nil, apierrors.New(
			apierrors.ValidationError,
			fmt.Sprintf("Reason is required when changing status to %s", req.Status),
		)
	}

	// Update status within a transaction
	var college *models.College
	err = s.db.WithPlatformTransaction(ctx, func(ctx context.Context, _ pgx.Tx) error {
		var txErr error
		college, txErr = s.repo.UpdateCollegeStatus(ctx, collegeID, req.Status, req.Reason)
		return txErr
	})

	if err != nil {
		return nil, apierrors.New(
			apierrors.InternalError,
			"Failed to update college status",
		)
	}

	return college, nil
}

// isValidCollegeCode checks if the college code contains only valid characters.
func isValidCollegeCode(code string) bool {
	if code == "" || len(code) > 100 {
		return false
	}
	for _, c := range code {
		if !((c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') ||
			(c >= '0' && c <= '9') || c == '-' || c == '_') {
			return false
		}
	}
	return true
}

// isValidTimezone performs basic timezone validation.
func isValidTimezone(tz string) bool {
	if tz == "" || len(tz) > 100 {
		return false
	}
	// Basic check: should contain at least one slash or be UTC
	return strings.Contains(tz, "/") || tz == "UTC"
}

// isValidStatusTransition validates college status transitions.
func isValidStatusTransition(currentStatus, newStatus string) bool {
	// Define allowed transitions
	transitions := map[string][]string{
		"ONBOARDING":  {"ACTIVE", "DEACTIVATED"},
		"ACTIVE":      {"SUSPENDED", "DEACTIVATED"},
		"SUSPENDED":   {"ACTIVE", "DEACTIVATED"},
		"DEACTIVATED": {"ACTIVE"}, // Allow reactivation
	}

	allowedTransitions, exists := transitions[currentStatus]
	if !exists {
		return false
	}

	// Allow same status (no-op)
	if currentStatus == newStatus {
		return true
	}

	for _, allowed := range allowedTransitions {
		if newStatus == allowed {
			return true
		}
	}

	return false
}

// requiresReason returns true if a status change requires a reason.
func requiresReason(status string) bool {
	return status == "SUSPENDED" || status == "DEACTIVATED"
}

// DBTX represents a database transaction or connection.
type DBTX interface {
	Exec(
		context.Context,
		string,
		...interface{},
	) (pgconn.CommandTag, error)

	Query(
		context.Context,
		string,
		...interface{},
	) (pgx.Rows, error)

	QueryRow(
		context.Context,
		string,
		...interface{},
	) pgx.Row
}

// Repository provides data access methods for the Super Admin module.
type Repository struct {
	Pool *pgxpool.Pool
}

// New creates a new Super Admin repository.
func New(db *database.Database) *Repository {
	return &Repository{
		Pool: db.Pool,
	}
}

// Executor returns the appropriate database executor (transaction or pool).
func (r *Repository) Executor(ctx context.Context) DBTX {
	if tx, ok := database.TxFromContext(ctx); ok {
		return tx
	}

	return r.Pool
}

// ListColleges retrieves a list of colleges with optional filtering and pagination.
func (r *Repository) ListColleges(
	ctx context.Context,
	status *string,
	limit, offset int,
) ([]models.College, int, error) {
	// Build the WHERE clause based on optional status filter
	whereClause := ""
	args := []interface{}{}
	argPos := 1

	if status != nil && *status != "" {
		whereClause = "WHERE status = $1"
		args = append(args, *status)
		argPos++
	}

	// Count total records
	countQuery := fmt.Sprintf(`
		SELECT COUNT(*)
		FROM platform.colleges
		%s
	`, whereClause)

	var total int
	err := r.Executor(ctx).QueryRow(ctx, countQuery, args...).Scan(&total)
	if err != nil {
		return nil, 0, fmt.Errorf("failed to count colleges: %w", err)
	}

	// Query for paginated results
	query := fmt.Sprintf(`
		SELECT
			college_id,
			college_code,
			name,
			city,
			state,
			timezone,
			status,
			contact_email,
			activated_at,
			suspended_at,
			deactivated_at,
			archived_at,
			lifecycle_reason,
			created_at,
			updated_at
		FROM platform.colleges
		%s
		ORDER BY name ASC
		LIMIT $%d OFFSET $%d
	`, whereClause, argPos, argPos+1)

	args = append(args, limit, offset)

	rows, err := r.Executor(ctx).Query(ctx, query, args...)
	if err != nil {
		return nil, 0, fmt.Errorf("failed to list colleges: %w", err)
	}
	defer rows.Close()

	var colleges []models.College
	for rows.Next() {
		var college models.College
		if err := rows.Scan(
			&college.CollegeID,
			&college.CollegeCode,
			&college.Name,
			&college.City,
			&college.State,
			&college.Timezone,
			&college.Status,
			&college.ContactEmail,
			&college.ActivatedAt,
			&college.SuspendedAt,
			&college.DeactivatedAt,
			&college.ArchivedAt,
			&college.LifecycleReason,
			&college.CreatedAt,
			&college.UpdatedAt,
		); err != nil {
			return nil, 0, fmt.Errorf("failed to scan college: %w", err)
		}
		colleges = append(colleges, college)
	}

	if err := rows.Err(); err != nil {
		return nil, 0, fmt.Errorf("failed while reading colleges: %w", err)
	}

	return colleges, total, nil
}

// CreateInitialCollegeOnboardingSteps creates the initial checklist.
// Call this inside the same transaction that creates the college.
func (r *Repository) CreateInitialCollegeOnboardingSteps(
	ctx context.Context,
	collegeID uuid.UUID,
) error {
	if collegeID == uuid.Nil {
		return fmt.Errorf("college ID is required for onboarding")
	}

	stepCodes := []string{
		"COLLEGE_PROFILE",
		"COLLEGE_ADMIN",
		"BRANDING",
		"ENTITLEMENTS",
		"READINESS_REVIEW",
		"GO_LIVE",
	}

	for _, stepCode := range stepCodes {
		_, err := r.Executor(ctx).Exec(
			ctx,
			`
			INSERT INTO platform.college_onboarding (
				college_id,
				step_code,
				status
			)
			VALUES ($1, $2, 'PENDING')
			ON CONFLICT (college_id, step_code) DO NOTHING
			`,
			collegeID,
			stepCode,
		)
		if err != nil {
			return fmt.Errorf(
				"create onboarding step %s: %w",
				stepCode,
				err,
			)
		}
	}

	return nil
}

// ListCollegeOnboardingSteps retrieves a college's onboarding checklist.
func (r *Repository) ListCollegeOnboardingSteps(
	ctx context.Context,
	collegeID uuid.UUID,
) ([]CollegeOnboardingStep, error) {
	rows, err := r.Executor(ctx).Query(
		ctx,
		`
		SELECT
			onboarding_id,
			college_id,
			step_code,
			status,
			completed_by,
			completed_at::text,
			notes
		FROM platform.college_onboarding
		WHERE college_id = $1
		ORDER BY created_at ASC, step_code ASC
		`,
		collegeID,
	)
	if err != nil {
		return nil, fmt.Errorf(
			"list college onboarding steps: %w",
			err,
		)
	}
	defer rows.Close()

	steps := make([]CollegeOnboardingStep, 0)

	for rows.Next() {
		var step CollegeOnboardingStep

		if err := rows.Scan(
			&step.OnboardingID,
			&step.CollegeID,
			&step.StepCode,
			&step.Status,
			&step.CompletedBy,
			&step.CompletedAt,
			&step.Notes,
		); err != nil {
			return nil, fmt.Errorf(
				"scan college onboarding step: %w",
				err,
			)
		}

		steps = append(steps, step)
	}

	if err := rows.Err(); err != nil {
		return nil, fmt.Errorf(
			"read college onboarding steps: %w",
			err,
		)
	}

	return steps, nil
}

type UpdateCollegeOnboardingParams struct {
	StepCode    string
	Status      string
	Notes       *string
	CompletedBy *uuid.UUID
}

type CollegeOnboardingUpdateResult struct {
	Step      CollegeOnboardingStep
	OldStatus string
}

// UpdateCollegeOnboardingStep updates a step inside the caller's transaction.
func (r *Repository) UpdateCollegeOnboardingStep(
	ctx context.Context,
	collegeID uuid.UUID,
	params UpdateCollegeOnboardingParams,
) (CollegeOnboardingUpdateResult, error) {
	if collegeID == uuid.Nil {
		return CollegeOnboardingUpdateResult{}, fmt.Errorf("college ID is required")
	}

	var result CollegeOnboardingUpdateResult

	err := r.Executor(ctx).QueryRow(ctx, `
		SELECT status
		FROM platform.college_onboarding
		WHERE college_id = $1 AND step_code = $2
		FOR UPDATE
	`, collegeID, params.StepCode).Scan(&result.OldStatus)

	if err != nil {
		if errors.Is(err, pgx.ErrNoRows) {
			return result, ErrNotFound
		}
		return result, fmt.Errorf("lock onboarding step: %w", err)
	}

	_, err = r.Executor(ctx).Exec(ctx, `
		UPDATE platform.college_onboarding
		SET
			status = $3,
			notes = $4,
			completed_by = CASE
				WHEN $3 = 'COMPLETED' THEN $5
				ELSE NULL
			END,
			completed_at = CASE
				WHEN $3 = 'COMPLETED' THEN NOW()
				ELSE NULL
			END,
			updated_at = NOW()
		WHERE college_id = $1 AND step_code = $2
	`, collegeID, params.StepCode, params.Status, params.Notes, params.CompletedBy)

	if err != nil {
		return result, fmt.Errorf("update onboarding step: %w", err)
	}

	var step CollegeOnboardingStep

	err = r.Executor(ctx).QueryRow(ctx, `
		SELECT
			onboarding_id, college_id, step_code, status,
			completed_by, completed_at::text, notes
		FROM platform.college_onboarding
		WHERE college_id = $1 AND step_code = $2
	`, collegeID, params.StepCode).Scan(
		&step.OnboardingID,
		&step.CollegeID,
		&step.StepCode,
		&step.Status,
		&step.CompletedBy,
		&step.CompletedAt,
		&step.Notes,
	)

	if err != nil {
		return result, fmt.Errorf("read updated onboarding step: %w", err)
	}

	result.Step = step
	return result, nil
}

// ValidateCollegeOnboardingReadyForActivation ensures all required
// onboarding steps exist and are completed before college activation.
// Call inside the same transaction as the college status update.
func (r *Repository) ValidateCollegeOnboardingReadyForActivation(
	ctx context.Context,
	collegeID uuid.UUID,
) error {
	rows, err := r.Executor(ctx).Query(
		ctx,
		`
		SELECT step_code, status
		FROM platform.college_onboarding
		WHERE college_id = $1
		ORDER BY step_code
		FOR UPDATE
		`,
		collegeID,
	)
	if err != nil {
		return fmt.Errorf("lock college onboarding checklist: %w", err)
	}
	defer rows.Close()

	requiredSteps := map[string]bool{
		"COLLEGE_PROFILE":  false,
		"COLLEGE_ADMIN":    false,
		"BRANDING":         false,
		"ENTITLEMENTS":     false,
		"READINESS_REVIEW": false,
		"GO_LIVE":          false,
	}

	for rows.Next() {
		var stepCode, status string
		if err := rows.Scan(&stepCode, &status); err != nil {
			return fmt.Errorf("read onboarding checklist: %w", err)
		}

		if _, required := requiredSteps[stepCode]; !required {
			continue
		}

		if status != "COMPLETED" {
			return fmt.Errorf(
				"onboarding step %s must be COMPLETED before activation",
				stepCode,
			)
		}

		requiredSteps[stepCode] = true
	}

	if err := rows.Err(); err != nil {
		return fmt.Errorf("iterate onboarding checklist: %w", err)
	}

	for stepCode, completed := range requiredSteps {
		if !completed {
			return fmt.Errorf(
				"required onboarding step %s is missing",
				stepCode,
			)
		}
	}

	return nil
}

type PlatformCollegeStatus struct {
	CollegeID       uuid.UUID `json:"college_id"`
	CollegeCode     string    `json:"college_code"`
	Name            string    `json:"name"`
	Status          string    `json:"status"`
	Timezone        string    `json:"timezone"`
	ActivatedAt     *string   `json:"activated_at,omitempty"`
	SuspendedAt     *string   `json:"suspended_at,omitempty"`
	DeactivatedAt   *string   `json:"deactivated_at,omitempty"`
	ArchivedAt      *string   `json:"archived_at,omitempty"`
	LifecycleReason *string   `json:"lifecycle_reason,omitempty"`
}

// GetCollegeStatusForUpdate reads the current status while locking the
// college row until the surrounding transaction commits or rolls back.
func (r *Repository) GetCollegeStatusForUpdate(
	ctx context.Context,
	collegeID uuid.UUID,
) (string, error) {
	var status string

	err := r.Executor(ctx).QueryRow(
		ctx,
		`
		SELECT status
		FROM platform.colleges
		WHERE college_id = $1
		FOR UPDATE
		`,
		collegeID,
	).Scan(&status)

	if err != nil {
		if errors.Is(err, pgx.ErrNoRows) {
			return "", ErrNotFound
		}
		return "", fmt.Errorf("failed to read college status: %w", err)
	}

	return status, nil
}

func (r *Repository) UpdateCollegeStatus(
	ctx context.Context,
	collegeID uuid.UUID,
	params UpdateCollegeStatusParams,
) (PlatformCollegeStatus, error) {
	executor := r.Executor(ctx)

	var college PlatformCollegeStatus

	err := executor.QueryRow(
		ctx,
		`
		UPDATE platform.colleges
		SET
			status = $1::text,
			lifecycle_reason = $2,

			activated_at = CASE
				WHEN $1::text = 'ACTIVE' AND status <> 'ACTIVE'
					THEN NOW()
				ELSE activated_at
			END,

			suspended_at = CASE
				WHEN $1::text = 'SUSPENDED' AND status <> 'SUSPENDED'
					THEN NOW()
				ELSE suspended_at
			END,

			deactivated_at = CASE
				WHEN $1::text = 'DEACTIVATED' AND status <> 'DEACTIVATED'
					THEN NOW()
				ELSE deactivated_at
			END,

			updated_at = NOW()

		WHERE college_id = $3

		RETURNING
			college_id,
			college_code,
			name,
			status,
			timezone,
			to_char(activated_at, 'YYYY-MM-DD"T"HH24:MI:SS"Z"'),
			to_char(suspended_at, 'YYYY-MM-DD"T"HH24:MI:SS"Z"'),
			to_char(deactivated_at, 'YYYY-MM-DD"T"HH24:MI:SS"Z"'),
			to_char(archived_at, 'YYYY-MM-DD"T"HH24:MI:SS"Z"'),
			lifecycle_reason
		`,
		params.Status,
		params.LifecycleReason,
		collegeID,
	).Scan(
		&college.CollegeID,
		&college.CollegeCode,
		&college.Name,
		&college.Status,
		&college.Timezone,
		&college.ActivatedAt,
		&college.SuspendedAt,
		&college.DeactivatedAt,
		&college.ArchivedAt,
		&college.LifecycleReason,
	)

	if err != nil {
		if errors.Is(err, pgx.ErrNoRows) {
			return PlatformCollegeStatus{}, ErrNotFound
		}

		var pgErr *pgconn.PgError

		if errors.As(err, &pgErr) {
			return PlatformCollegeStatus{}, fmt.Errorf(
				"failed to update college status: %w",
				err,
			)
		}

		return PlatformCollegeStatus{}, fmt.Errorf(
			"failed to update college status: %w",
			err,
		)
	}

	return college, nil
}
